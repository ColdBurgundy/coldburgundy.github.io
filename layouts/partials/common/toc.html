  • 로 남깁니다. 그래서 수식이 든 소제목은 render-heading 이 달아 둔 data-label(common/heading-label.html, 예: "복잡도 O(n²) 분석")로 글자를 바꾸고, 빈 자리는 같은 깊이의 그 소제목으로 채웁니다. (영어 번역
    속 소제목은 Hugo 목차에 없으므로 건너뜁니다.) 사용: {{ $toc := partial "common/toc.html" . }} */ -}} {{- $toc := string .TableOfContents -}} {{- if strings.Contains (string .Content) "data-label=" -}} {{- $c := replaceRE `(?s)` "" (string .Content) -}} {{- $heads := slice -}} {{- range findRESubmatch `]*)>` $c -}} {{- $label := "" -}} {{- with findRESubmatch `\sdata-label="([^"]*)"` (index . 3) 1 }}{{ $label = index (index . 0) 1 }}{{ end -}} {{- $heads = $heads | append (dict "level" (int (index . 1)) "id" (index . 2) "label" $label) -}} {{- end -}} {{- /* 1) Hugo 가 앵커를 만든 소제목(목차 링크)과, 소제목 단계 =
      깊이 + offset (startLevel 2 면 1) */ -}} {{- $linked := slice -}} {{- range findRE `` $toc }}{{ $linked = $linked | append (strings.TrimSuffix `">` (strings.TrimPrefix `` $toc -}} {{- if eq . "
        " }}{{ $depth = add $depth 1 }} {{- else if eq . "
      " }}{{ $depth = sub $depth 1 }} {{- else -}} {{- $id := strings.TrimSuffix `">` (strings.TrimPrefix `      ).*?()` (replaceRE `([\.+*?()|\[\]{}^$])` `\$1` .id) -}} {{- $toc = replaceRE $re (printf "${1}%s${2}" (replace .label "$" "$$")) $toc -}} {{- end -}} {{- end -}} {{- /* 3) 수식만 있는 소제목: 링크 없는 자리(
    • , 아래 단계를 품은
      • )를 바로 앞 · 뒤 목차 링크 사이에 있는 같은 단계의 수식 소제목으로 채웁니다. 못 찾으면 빈
      • 는 지우고, 단계를 건너뛴 자리(
        • )는 그대로 둡니다. */ -}} {{- $keep := "\uE000" -}} {{- $depth = 0 -}} {{- $pos := 0 -}} {{- range findRE `
            |
          |
        • (?:
          • |\s*
            )|` $toc -}} {{- if eq . "
              " }}{{ $depth = add $depth 1 }} {{- else if eq . "
            " }}{{ $depth = sub $depth 1 }} {{- else if hasPrefix . "` (strings.TrimPrefix `            %s` $h.id $h.label -}} {{- $pos = add $k 1 -}} {{- break -}} {{- end -}} {{- end -}} {{- $new := replace . "
          • " (print "") 1 -}} {{- if $fill }}{{ $new = replace . "
          • " (print "
          • " $fill) 1 }} {{- else if eq . "
          • " }}{{ $new = "" }}{{ end -}} {{- $toc = replace $toc . $new 1 -}} {{- if hasSuffix . "
              " }}{{ $depth = add $depth 1 }}{{ end -}} {{- end -}} {{- end -}} {{- $toc = replace $toc $keep "" -}} {{- /* 비워진
                ·
              • 정리 (목차 단계 밖의 수식 소제목이 남긴 껍데기) */ -}} {{- range seq 3 }}{{ $toc = replaceRE `
              • \s*
                • |
                  \s*
                ` "" $toc }}{{ end -}} {{- end -}} {{- return ($toc | safeHTML) -}}
