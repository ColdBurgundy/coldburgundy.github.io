{{- /* 소제목 HTML(render-heading 의 .Text) → 평문 라벨. 예: "복잡도 $O(n^2)$ 분석" → "복잡도 O(n²) 분석" KaTeX 출력은 MathML · TeX 주석 · 화면용 HTML 을 함께 담고 있어 태그만 지우면 글자가 세 번 나오므로, 수식 하나를 통째로 TeX 주석 → common/tex-plain.html 로 다듬은 글자 하나로 바꿉니다. */ -}} {{- $html := string . -}} {{- $mark := "\uE000" -}} {{- /* 수식의 MathML 부분(TeX 주석 포함)을 "다듬은 글자 + 표시"로 바꾸고, 뒤따르는 화면용 HTML 은 아래에서 통째로 버립니다 */ -}} {{- range findRESubmatch `(?s).*?(.*?).*?` $html -}} {{- $t := partial "common/tex-plain.html" (index . 1 | htmlUnescape) -}} {{- $html = replace $html (index . 0) (printf "%s%s" (htmlEscape $t) $mark) -}} {{- end -}} {{- $label := "" -}} {{- $skip := -1 -}} {{- range findRE (printf `%s|]*>||<[^>]*>|[^<%s]+` $mark $mark) $html -}} {{- if eq . $mark }}{{ $skip = 0 }}{{ continue }}{{ end -}} {{- if ge $skip 0 -}} {{- /* 화면용 HTML(.katex-html) 과 수식을 닫는 까지 건너뜀 */ -}} {{- if hasPrefix . "" }}{{ $skip = sub $skip 1 }}{{ if lt $skip 0 }}{{ $skip = -1 }}{{ end }} {{- end -}} {{- continue -}} {{- end -}} {{- $label = print $label . -}} {{- end -}} {{- return ($label | plainify | htmlUnescape | replaceRE `\s+` " " | strings.TrimSpace) -}}
